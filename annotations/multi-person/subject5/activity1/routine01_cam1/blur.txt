import json
import cv2
import numpy as np
from pathlib import Path

def load_mask_from_json(json_path, labels=None):
    p = Path(json_path)
    data = json.loads(p.read_text())

    H = int(data["imageHeight"])
    W = int(data["imageWidth"])

    mask = np.zeros((H, W), dtype=np.uint8)

    for shp in data["shapes"]:
        if shp["shape_type"] != "polygon":
            continue

        lbl = shp["label"]
        if labels and lbl not in labels:
            continue

        pts = np.array(shp["points"], dtype=np.int32)
        cv2.fillPoly(mask, [pts], 255)

    return mask, data.get("imagePath")


def bbox_from_mask(mask):
    ys, xs = np.where(mask > 0)
    if len(xs) == 0:
        return None
    return xs.min(), ys.min(), xs.max(), ys.max()


def expand_mask_up(mask, pixels=10):
    """
    Expande máscara apenas para cima
    """
    kernel = np.ones((pixels, 1), np.uint8)
    return cv2.dilate(mask, kernel, anchor=(0, pixels-1))


def blur_roi(img, mask, ksize=51, feather=15):
    """
    Blur apenas dentro da máscara
    """
    blurred = cv2.GaussianBlur(img, (ksize, ksize), 0)

    m = (mask > 0).astype(np.float32)
    m = cv2.GaussianBlur(m, (0,0), feather)
    m = np.clip(m, 0, 1)

    m = m[...,None]

    out = img*(1-m) + blurred*m
    return out.astype(np.uint8)


def darken_bottom_edge(img, mask, band_px=15, strength=0.4):
    """
    escurece levemente a borda inferior da máscara
    simulando sombra/fim da camisa
    """
    ys, xs = np.where(mask > 0)
    if len(xs) == 0:
        return img

    bottom = ys.max()

    y0 = max(0, bottom-band_px)
    y1 = bottom

    alpha = np.zeros(mask.shape, dtype=np.float32)

    for y in range(y0, y1):
        t = (y-y0)/(y1-y0)
        alpha[y,:] = t*strength

    alpha[mask==0] = 0

    out = img.astype(np.float32)
    out *= (1-alpha[...,None])

    return np.clip(out,0,255).astype(np.uint8)


def process(json_path):

    mask, img_path = load_mask_from_json(json_path)

    jp = Path(json_path)

    if img_path:
        frame_path = jp.parent / img_path
    else:
        frame_path = jp.with_suffix(".jpg")

    img = cv2.imread(str(frame_path))

    # expandir máscara para cima
    mask2 = expand_mask_up(mask, pixels=12)

    # blur na região
    out = blur_roi(img, mask2)

    # sombra leve na borda
    out = darken_bottom_edge(out, mask2)

    out_path = jp.with_suffix("").as_posix() + "_blur.jpg"

    cv2.imwrite(out_path, out)

    print("OK ->", out_path)


if __name__ == "__main__":
    process("frame_crop_cam1_000300.json")
